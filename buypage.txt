"use client";

import { useEffect, useState } from "react";

export default function BuyNumberPage() {

const [countries, setCountries] =
  useState<any[]>([]);

  const [services, setServices] =
    useState<any>({});

  const [selectedCountry, setSelectedCountry] =
    useState("");

  const [loading, setLoading] =
    useState(true);

  const [loadingServices, setLoadingServices] =
    useState(false);

  const dollarRate = 1600;

  const profitMargin = 1.18;

  useEffect(() => {

    const fetchCountries = async () => {

      try {

        const response = await fetch(
          "/api/countries"
        );

        const data = await response.json();

        setCountries(data);

      } catch (error) {

        console.log(error);

      } finally {

        setLoading(false);

      }

    };

    fetchCountries();

  }, []);

  const fetchServices = async (
    country: string
  ) => {

    try {

      setLoadingServices(true);

      setSelectedCountry(country);

      const response = await fetch(
        `https://5sim.net/v1/guest/products/${country}/any`
      );

      const data = await response.json();

      setServices(data);

    } catch (error) {

      console.log(error);

    } finally {

      setLoadingServices(false);

    }

  };

  return (
    <main className="min-h-screen bg-gray-100 p-6">

      <h1 className="text-4xl font-bold text-blue-600 mb-8">
        Buy OTP Number
      </h1>

      {!selectedCountry ? (

        <>

          <h2 className="text-2xl font-bold mb-5">
            Select Country
          </h2>

          {loading ? (

            <p className="text-lg font-semibold">
              Loading countries...
            </p>

          ) : (

<div className="grid gap-4">{countries.map((country: any) => (

<button
  key={country.ID}
  onClick={() =>
    fetchServices(country.ID)
  }
  className="bg-white p-5 rounded-3xl shadow-sm text-left font-bold text-xl"
>
  {country.name}

</button>

))}

</div>
                  <button
                    key={country}
                    onClick={() =>
                      fetchServices(country)
                    }
                    className="bg-white p-5 rounded-3xl shadow-sm text-left font-bold text-xl"
                  >
                    {country}
                  </button>

                )
              )}

            </div>

          )}

        </>

      ) : (

        <>

          <button
            onClick={() => {

              setSelectedCountry("");

              setServices({});

            }}
            className="mb-5 bg-black text-white px-5 py-3 rounded-2xl"
          >
            Back
          </button>

          <h2 className="text-3xl font-bold mb-5 capitalize">
            {selectedCountry} Services
          </h2>

          {loadingServices ? (

            <p className="text-lg font-semibold">
              Loading services...
            </p>

          ) : (

            <div className="grid gap-4">

              {Object.entries(services).map(
                ([service, details]: any) => {

                  const usdPrice =
                    details?.Price || 0;

                  const nairaPrice =
                    Math.ceil(
                      usdPrice *
                      dollarRate *
                      profitMargin
                    );

                  return (

                    <div
                      key={service}
                      className="bg-white p-5 rounded-3xl shadow-sm"
                    >

                      <h3 className="text-2xl font-bold capitalize">
                        {service}
                      </h3>

                      <p className="text-green-600 font-bold text-2xl mt-3">
                        ₦{nairaPrice.toLocaleString()}
                      </p>

                      <button
                        onClick={() =>
                          alert(
                            "Insufficient Balance"
                          )
                        }
                        className="mt-5 bg-blue-600 text-white py-4 rounded-2xl w-full font-bold"
                      >
                        Proceed
                      </button>

                    </div>

                  );

                }
              )}

            </div>

          )}

        </>

      )}

    </main>
  );
}
